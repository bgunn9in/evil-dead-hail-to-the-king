if(EXISTS "${TEST_ASSETS}/dreamcast-runtime.json")
    if(NOT EXISTS "${NATIVE_FIXTURE_ASSETS}/dreamcast-runtime.json")
        message(FATAL_ERROR "Prepare the immutable original Dreamcast 1x fixture profile before configuring.")
    endif()
    file(SHA256 "${NATIVE_FIXTURE_ASSETS}/dreamcast-runtime.json" NATIVE_FIXTURE_SEAL)
    if(NOT NATIVE_FIXTURE_SEAL STREQUAL "7e957354cb92267f60f2d6f2e18ce066a4194139f4e7b8ac6ff2bc6c44ed985d")
        message(FATAL_ERROR "The original Dreamcast 1x fixture seal is unknown.")
    endif()
    file(READ "${TEST_ASSETS}/dreamcast-runtime.json" DREAMCAST_PROFILE)
    string(JSON DREAMCAST_FORMAT GET "${DREAMCAST_PROFILE}" format)
    string(JSON DREAMCAST_STATUS GET "${DREAMCAST_PROFILE}" status)
    if(NOT DREAMCAST_FORMAT STREQUAL "ED-DREAMCAST-RUNTIME1" OR NOT DREAMCAST_STATUS STREQUAL "verified-import")
        message(FATAL_ERROR "Only a sealed verified Dreamcast resource profile is supported.")
    endif()
else()
    message(STATUS "Source-only build: gameplay and asset-dependent tests require a local sealed Dreamcast profile.")
endif()

foreach(STAGE vert frag bgfrag hudfrag particlefrag opaquefrag maskfrag postfrag)
    string(JSON EXPECTED_SPIRV_SHA GET "${SPIRV_MANIFEST}" shaders ${SHADER_INDEX} sha256)
    set(PREBUILT_SPIRV "${CMAKE_CURRENT_SOURCE_DIR}/runtime/shaders/scene.${STAGE}.spv")
    file(SHA256 "${PREBUILT_SPIRV}" CURRENT_SPIRV_SHA)
    if(NOT EXPECTED_SPIRV_SHA STREQUAL CURRENT_SPIRV_SHA)
        message(FATAL_ERROR "SPIR-V checksum mismatch: ${PREBUILT_SPIRV}")
    endif()
    configure_file("${PREBUILT_SPIRV}" "${CMAKE_BINARY_DIR}/shaders/scene.${STAGE}.spv" COPYONLY)
    list(APPEND SHADERS "${CMAKE_BINARY_DIR}/shaders/scene.${STAGE}.spv")
    math(EXPR SHADER_INDEX "${SHADER_INDEX}+1")
endforeach()
